### 3.2 Prisma Schema 定义

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id              Int                @id @default(autoincrement())
  username        String             @unique @db.VarChar(20)
  phone           String             @unique @db.VarChar(20)
  passwordHash    String             @map("password_hash") @db.VarChar(255)
  role            String             @default("user") @db.VarChar(10)
  createdAt       DateTime           @default(now()) @map("created_at")
  form1Submissions Form1Submission[]
  form2Submissions Form2Submission[]

  @@map("users")
}

model Form1Submission {
  id                Int      @id @default(autoincrement())
  userId            Int      @map("user_id")
  name              String   @db.VarChar(50)
  phone             String   @db.VarChar(20)
  consultationType  String   @map("consultation_type") @db.VarChar(20)
  preferredTime     String   @map("preferred_time") @db.VarChar(50)
  brief             String   @db.Text
  submittedAt       DateTime @default(now()) @map("submitted_at")
  versionNumber     Int      @map("version_number")
  feishuRecordId    String?  @map("feishu_record_id") @db.VarChar(100)
  feishuSyncStatus  String   @default("pending") @map("feishu_sync_status") @db.VarChar(20)
  user              User     @relation(fields: [userId], references: [id])

  @@map("form1_submissions")
}

model Form2Submission {
  id                Int      @id @default(autoincrement())
  userId            Int      @map("user_id")
  formData          Json     @map("form_data") @db.JsonB
  submittedAt       DateTime @default(now()) @map("submitted_at")
  versionNumber     Int      @map("version_number")
  feishuRecordId    String?  @map("feishu_record_id") @db.VarChar(100)
  feishuSyncStatus  String   @default("pending") @map("feishu_sync_status") @db.VarChar(20)
  user              User     @relation(fields: [userId], references: [id])

  @@map("form2_submissions")
}
```

#### JWT Token 结构

```typescript
interface JwtPayload {
  userId: number;
  phone: string;
  role: 'user' | 'admin';
  iat: number;  // 签发时间
  exp: number;  // 过期时间（7天后）
```

#### 认证中间件流程

```

  ↓
检查 Authorization Header
  ↓ (存在)
提取 Bearer Token
  ↓
验证 JWT 签名及过期时间
  ↓ (有效)
将 JwtPayload 注入 req.user
  ↓
继续处理请求
```

#### 管理员守卫中间件

```typescript
// 在 auth 中间件之后执行
const adminGuard = (req: Request, res: Response, next: NextFunction) => {
  if (req.user?.role !== 'admin') {
    return res.status(403).json({
      code: 403,
      message: '无权限访问',
    });
  }
  next();
};
```

### 5.2 表单提交流程

```
用户提交表单
  ↓
JWT 认证校验
  ↓
请求参数校验
  ↓
计算版本号（查询同用户同表单最大版本号 + 1）
  ↓
写入数据库（sync_status = 'pending'）
  ↓
返回成功响应给用户  ← ← ← ← ← 不等待飞书同步
  ↓
异步触发飞书同步
  ↓
┌─ 成功 → 更新 sync_status = 'success'，记录 feishu_record_id
└─ 失败 → 更新 sync_status = 'failed'，记录错误日志
```

#### 版本号计算逻辑

```typescript
const getNextVersion = async (userId: number, formType: 'form1' | 'form2'): Promise<number> => {
  const model = formType === 'form1' ? prisma.form1Submission : prisma.form2Submission;
  const lastSubmission = await model.findFirst({
    where: { userId },
    orderBy: { versionNumber: 'desc' },
    select: { versionNumber: true },
  });
  return (lastSubmission?.versionNumber ?? 0) + 1;
};
```

### 5.3 飞书同步模块

#### Token 管理

飞书 API 需要 `tenant_access_token`，有效期 2 小时，需缓存并自动续期。

```typescript
class FeishuTokenManager {
  private token: string | null = null;
  private expiresAt: number = 0;

  async getToken(): Promise<string> {
    if (this.token && Date.now() < this.expiresAt) {
      return this.token;
    }
    const response = await axios.post(
      'https://open.feishu.cn/open-apis/auth/v3/tenant_access_token/internal',
      {
        app_id: process.env.FEISHU_APP_ID,
        app_secret: process.env.FEISHU_APP_SECRET,
      }
    );
    this.token = response.data.tenant_access_token;
    // 提前 5 分钟续期
    this.expiresAt = Date.now() + (response.data.expire - 300) * 1000;
    return this.token!;
  }
}
```

#### 写入记录

```typescript
const createRecord = async (
  appToken: string,
  tableId: string,
  fields: Record<string, unknown>
): Promise<string> => {
  const token = await feishuTokenManager.getToken();
  const response = await axios.post(
    `https://open.feishu.cn/open-apis/bitable/v1/apps/${appToken}/tables/${tableId}/records`,
    { fields },
    {
      headers: {
        Authorization: `Bearer ${token}`,
        'Content-Type': 'application/json',
      },
    }
  );
  return response.data.data.record.record_id;
};
```

```typescript
    '提交时间': submission.submittedAt.getTime(),
    '版本号': submission.versionNumber,
    '完整数据': JSON.stringify(formData),
  };
};
```

### 5.4 统一响应格式

```typescript
interface ApiResponse<T = unknown> {
  code: number;       // 0 = 成功，非 0 = 错误
  message: string;    // 人类可读的描述
  data?: T;           // 业务数据
}

// 成功响应
const success = <T>(data: T, message = '操作成功'): ApiResponse<T> => ({
  code: 0,
  message,
  data,
});

// 错误响应
const error = (code: number, message: string): ApiResponse => ({
  code,
  message,
});
```

### 5.5 错误码定义

| 错误码 | 说明 |
|--------|------|
| 0 | 成功 |
| 400 | 请求参数错误 |
| 401 | 未认证（Token 缺失或无效） |
| 403 | 无权限（非管理员） |
| 404 | 资源不存在 |
| 409 | 冲突（如手机号已注册） |
| 500 | 服务器内部错误 |
| 1001 | 验证码错误 |
| 1002 | 用户不存在 |
| 2001 | 飞书同步失败 |

```typescript
const AuthGuard: React.FC<{ children: React.ReactNode }> = ({ children }) => {
```

## 七、安全设计

### 7.1 认证安全

| 措施 | 说明 |
|------|------|
| 密码哈希 | bcrypt，salt rounds = 10 |
| JWT 签名 | HS256 算法，密钥从环境变量读取 |
| Token 过期 | 7 天有效期 |
| 请求头验证 | `Authorization: Bearer <token>` |

### 7.2 接口安全

| 措施 | 说明 |
|------|------|
| CORS | 仅允许指定域名 |
| Rate Limiting | 登录接口限频（如 5 次/分钟） |
| 输入校验 | 所有接口参数使用 zod 或 joi 校验 |
| SQL 注入防护 | Prisma ORM 参数化查询 |

### 7.3 敏感信息

| 措施 | 说明 |
|------|------|
| 环境变量 | 所有密钥通过 `.env` 管理，不提交至版本控制 |
| 密码 | 数据库仅存储哈希值，永不明文存储或传输 |
| 日志脱敏 | 日志中不记录密码、Token 等敏感字段 |

### 9.1 日志规范

|------|----------|
